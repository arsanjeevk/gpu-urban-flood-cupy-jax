\vspace{7mm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Scientific need}
Urban flooding requires conservation-based simulation of rapidly changing wet and dry flow
over complex topography and coupled drainage infrastructure.
\end{block}
\column{0.48\textwidth}
\begin{block}{Computational need}
Real-time or near-real-time use demands substantially faster execution without changing the
equations, forcing, numerical controls or physical interpretation.
\end{block}
\end{columns}

\vspace{2mm}
\centering\color{hfblue}\large\bfseries Acceleration matters only if numerical fidelity is preserved.
\end{frame}

% 3
\begin{frame}{2. Research question: can acceleration preserve the model?}
\begin{block}{Central research question}
Can the \textbf{same physics-based urban flood model} be implemented through two GPU
programming approaches while preserving numerical equivalence, and how do their measured
computational performances compare?
\end{block}

\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Objectives}
\begin{itemize}\tightitem
\item Implement equivalent GPU execution paths.
\item Hold physics, inputs and numerical controls fixed.
\item Test cross-backend parity and repeatability.
\item Compare synchronized GPU runtimes.
\end{itemize}
\column{0.48\textwidth}
\textbf{Evidence required}
\begin{enumerate}\tightitem
\item Scientific model and discretization
\item Controlled implementation correspondence
\item Predefined correctness gates
\item Performance only after parity passes
\end{enumerate}
\end{columns}

\vspace{3mm}
\centering\small Numerical verification is the prerequisite; observed-flood validation is a separate question.
\end{frame}

\begin{frame}{3. The scientific core is the two-dimensional shallow-water model}
\small
\begin{block}{Conservative balance law}
\[
\frac{\partial\bm U}{\partial t}+\frac{\partial\bm F}{\partial x}
+\frac{\partial\bm G}{\partial y}=\bm S,
\qquad \bm U=[h,hu,hv]^{\mathsf T}.
\]
\end{block}

\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\textbf{What the state represents}
\begin{itemize}\tightitem
\item $h$: water depth
\item $hu,hv$: depth-integrated momentum
\item Fluxes transport conserved water and momentum
\item Sources represent terrain and coupled processes
\end{itemize}
\column{0.49\textwidth}
\textbf{Why finite volume?}

The method balances fluxes across cell edges, directly supports conservation on an
unstructured triangular mesh, and accommodates irregular urban geometry, wet--dry fronts,
terrain steps and boundary interfaces.
\end{columns}

\vspace{3mm}
\centering\color{hfblue}\bfseries The physics defines what must be preserved by both GPU paths.
\end{frame}

% 5
\begin{frame}{4. The flood model is computed as a staged conservative update}
\centering
\begin{tikzpicture}[node distance=3mm]
  \node[flow,text width=21mm] (mesh) {Triangular mesh\\and cell state};
  \node[flow,text width=23mm,right=of mesh] (recon) {Hydrostatic\\reconstruction};
  \node[flow,text width=19mm,right=of recon] (hll) {HLL edge\\fluxes};
  \node[flow,text width=22mm,right=of hll] (acc) {Conservative\\accumulation};
  \node[flow,text width=22mm,right=of acc] (src) {Physical and\\network sources};
  \draw[flowarrow] (mesh) -- (recon);
  \draw[flowarrow] (recon) -- (hll);
  \draw[flowarrow] (hll) -- (acc);
  \draw[flowarrow] (acc) -- (src);
\end{tikzpicture}

\vspace{4mm}
\[
\bm U_i^{n+1}=\bm U_i^n-\frac{\Delta t}{A_i}
\sum_{e\in\partial i}L_e\widehat{\bm F}_{e,i}+\Delta t\,\bm S_i.
\]

\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\small
\textbf{Surface physics:} bed elevation, HLL flux, wetting/drying, semi-implicit Manning
friction, rainfall and Horton infiltration.
\column{0.49\textwidth}
\small
\textbf{Coupled processes:} drainage capture, link routing, surcharge, outfalls, recharge,
external inflow and outflow-only boundaries.
\end{columns}

\vspace{3mm}
\centering\small CFL-controlled adaptive timesteps keep the explicit update stable.
\end{frame}

\begin{frame}{5. The workload is highly parallel---but structurally difficult}
\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\centering
\begin{tikzpicture}[node distance=5mm]
  \node[flow,text width=44mm] (edges) {1,490,015 edge-local\\reconstruction and flux tasks};
  \node[flow,text width=44mm,below=of edges] (cells) {Indexed accumulation into\\981,880 triangular cells};
  \node[flow,text width=44mm,below=of cells] (network) {139,798 drainage links and\\118,768 drainage nodes};
  \draw[flowarrow] (edges) -- (cells);
  \draw[flowarrow] (cells) -- (network);
\end{tikzpicture}
\column{0.42\textwidth}
\textbf{GPU opportunity}
\begin{itemize}\tightitem
\item Many edge, cell and link operations are independent.
\item Major arrays can remain device-resident.
\end{itemize}

\vspace{3mm}
\textbf{GPU challenge}
\begin{itemize}\tightitem
\item Irregular mesh connectivity
\item Scatter/indexed reductions
\item Wet--dry branching and safeguards
\item Adaptive time integration
\item Surface--network exchange ordering
\end{itemize}
\end{columns}

\vspace{3mm}
\centering\color{hfblue}\bfseries This motivates two implementation strategies for one numerical model.
\end{frame}

\end{block}
